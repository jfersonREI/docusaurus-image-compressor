const fs = require("fs");
const path = require("path");

const input = path.resolve(__dirname, "../../pa11y-ci-report/results.json");
const output = path.resolve(__dirname, "../../pa11y-ci-report/issues.csv");

const data = JSON.parse(fs.readFileSync(input, "utf8"));
const results = data.results || {};

const rows = [["URL", "Type", "Code", "Message", "Selector", "Context"]];

for (const [url, issues] of Object.entries(results)) {
  if (!Array.isArray(issues) || issues.length === 0) continue;

  for (const issue of issues) {
    rows.push([
      url,
      issue.type || "",
      issue.code || "",
      `"${(issue.message || "").replace(/"/g, '""')}"`,
      `"${(issue.selector || "").replace(/"/g, '""')}"`,
      `"${(issue.context || "").replace(/"/g, '""')}"`,
    ]);
  }
}

const csv = rows.map((row) => row.join(",")).join("\n");
fs.writeFileSync(output, csv);

console.log(`Wrote ${rows.length - 1} issues to ${output}`);
