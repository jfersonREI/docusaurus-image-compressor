const fs = require("fs");
const path = require("path");
const ExcelJS = require("exceljs");

const input = path.resolve(__dirname, "../../pa11y-ci-report/results.json");
const output = path.resolve(__dirname, "../../pa11y-ci-report/issues.xlsx");

function cleanUrl(fullUrl) {
  try {
    const url = new URL(fullUrl);
    return url.pathname || fullUrl;
  } catch {
    return fullUrl;
  }
}

async function main() {
  const data = JSON.parse(fs.readFileSync(input, "utf8"));
  const results = data.results || {};

  const issues = [];

  for (const [url, pageIssues] of Object.entries(results)) {
    if (!Array.isArray(pageIssues) || pageIssues.length === 0) continue;

    const clean = cleanUrl(url);

    for (const issue of pageIssues) {
      issues.push({
        url: clean,
        fullUrl: url,
        type: issue.type || "",
        code: issue.code || "",
        message: issue.message || "",
      });
    }
  }

  // Sort by URL so issues for the same page are grouped
  issues.sort((a, b) => a.url.localeCompare(b.url));

  // Summary: group by code
  const summaryMap = new Map();
  for (const issue of issues) {
    const key = issue.code;
    if (!summaryMap.has(key)) {
      summaryMap.set(key, {
        code: issue.code,
        type: issue.type,
        message: issue.message,
        count: 0,
        exampleUrl: issue.url,
      });
    }
    summaryMap.get(key).count += 1;
  }

  const summary = Array.from(summaryMap.values()).sort(
    (a, b) => b.count - a.count,
  );

  const workbook = new ExcelJS.Workbook();

  // Summary sheet
  const summarySheet = workbook.addWorksheet("Summary");
  summarySheet.columns = [
    { header: "Count", key: "count", width: 10 },
    { header: "Type", key: "type", width: 12 },
    { header: "Code", key: "code", width: 50 },
    { header: "Message", key: "message", width: 80 },
    { header: "Example Path", key: "exampleUrl", width: 60 },
  ];
  summarySheet.addRows(summary);
  summarySheet.getRow(1).font = { bold: true };
  summarySheet.autoFilter = "A1:E1";
  summarySheet.views = [{ state: "frozen", ySplit: 1 }];
  summarySheet.getColumn("message").alignment = {
    wrapText: true,
    vertical: "top",
  };

  // Color summary by type
  summarySheet.eachRow((row, rowNumber) => {
    if (rowNumber === 1) return;
    const type = row.getCell("type").value;
    if (type === "error") {
      row.getCell("type").fill = {
        type: "pattern",
        pattern: "solid",
        fgColor: { argb: "FFFFC7CE" },
      };
    } else if (type === "warning") {
      row.getCell("type").fill = {
        type: "pattern",
        pattern: "solid",
        fgColor: { argb: "FFFFEB9C" },
      };
    }
  });

  // Issues sheet
  const issuesSheet = workbook.addWorksheet("Issues");
  issuesSheet.columns = [
    { header: "Path", key: "url", width: 60 },
    { header: "Type", key: "type", width: 12 },
    { header: "Code", key: "code", width: 50 },
    { header: "Message", key: "message", width: 80 },
  ];

  issues.forEach((issue) => {
    const row = issuesSheet.addRow({
      url: issue.url,
      type: issue.type,
      code: issue.code,
      message: issue.message,
    });

    // Make the path a clickable hyperlink
    row.getCell("url").value = {
      text: issue.url,
      hyperlink: issue.fullUrl,
    };
    row.getCell("url").font = { color: { argb: "FF0000FF" }, underline: true };

    // Color by type
    if (issue.type === "error") {
      row.getCell("type").fill = {
        type: "pattern",
        pattern: "solid",
        fgColor: { argb: "FFFFC7CE" },
      };
    } else if (issue.type === "warning") {
      row.getCell("type").fill = {
        type: "pattern",
        pattern: "solid",
        fgColor: { argb: "FFFFEB9C" },
      };
    }
  });

  issuesSheet.getRow(1).font = { bold: true };
  issuesSheet.autoFilter = "A1:D1";
  issuesSheet.views = [{ state: "frozen", ySplit: 1 }];
  issuesSheet.getColumn("message").alignment = {
    wrapText: true,
    vertical: "top",
  };

  await workbook.xlsx.writeFile(output);
  console.log(`Wrote ${issues.length} issues to ${output}`);
  console.log(`Summary has ${summary.length} unique issue types`);
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
