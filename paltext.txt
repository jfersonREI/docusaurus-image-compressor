import React from "react";
import { render, screen, fireEvent } from "@testing-library/react";
import HomepageSearch from "./index";

describe("HomepageSearch", () => {
  it("renders and submits the search form", () => {
    const { asFragment } = render(<HomepageSearch />);
    expect(asFragment()).toMatchSnapshot();

    const searchInput = screen.getByRole("searchbox");
    fireEvent.change(searchInput, { target: { value: "Vitest" } });
    expect(searchInput).toHaveValue("Vitest");

    // Prefer the button or a more specific form query
    const submitButton = screen.getByRole("button", { name: /search/i });
    fireEvent.click(submitButton);

    // Or, if you prefer to keep form submit:
    // const form = screen.getByRole('search'); // after adding role="search"
    // fireEvent.submit(form);
  });
});
