import React, { useState, FormEvent } from "react";
import { useHistory } from "@docusaurus/router";
import useBaseUrl from "@docusaurus/useBaseUrl";
import { Search } from "lucide-react";
import styles from "./styles.module.css";

export default function HomepageSearch(): React.JSX.Element {
  const [query, setQuery] = useState("");
  const history = useHistory();
  const searchPath = useBaseUrl("/search");

  const handleSubmit = (e: FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    if (query.trim()) {
      history.push(`${searchPath}?q=${encodeURIComponent(query.trim())}`);
    }
  };

  return (
    <div className={styles.homepageSearchContainer}>
      <form
        aria-label="Search documentation"
        onSubmit={handleSubmit}
        className={styles.homepageSearchForm}
        role="search"
      >
        <input
          type="search"
          className={styles.homepageSearchInput}
          placeholder="Search documentation"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          aria-label="Search documentation"
        />
        <button
          type="submit"
          className={styles.homepageSearchButton}
          aria-label="Search"
        >
          <Search size={18} aria-hidden="true" />
          <span className={styles.srOnly}>Search</span>
        </button>
      </form>
    </div>
  );
}
